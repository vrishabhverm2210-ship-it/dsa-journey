1class Solution {
2public:
3    int countSpecialIntegers(vector<int>& nums) {
4        unordered_map<int, vector<int>> mpp1;
5        int n=nums.size();
6        for(int i=0;i<n;i++){
7            mpp1[nums[i]].push_back(i);
8        }
9        int count=0;
10        for(auto i:mpp1){
11            if(i.second.size()!=3)continue;
12            else{
13                if(i.second[1]-i.second[0]==i.second[2]-i.second[1])count++;
14            }
15        }
16        return count;
17    }
18};
