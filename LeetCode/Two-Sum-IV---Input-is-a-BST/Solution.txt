1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14TreeNode* getsmaller(stack<TreeNode*>&asc,TreeNode*root){
15    TreeNode*  smaller=asc.top();
16    asc.pop();
17    TreeNode* righter=smaller->right;
18    while(righter){
19        asc.push(righter);
20        righter=righter->left;
21    }
22    return smaller;
23
24}
25TreeNode* getbigger(stack<TreeNode*>&desc,TreeNode*root){
26    TreeNode*  bigger=desc.top();
27    desc.pop();
28    TreeNode* lefter=bigger->left;
29    while(lefter){
30        desc.push(lefter);
31        lefter=lefter->right;
32    }
33    return bigger;
34
35}
36    bool findTarget(TreeNode* root, int k) {
37        // firstly build stack for storing the smallest elements
38        stack<TreeNode*>asc;
39        TreeNode* temp=root;
40        while(temp){
41            asc.push(temp);
42            temp=temp->left;
43        }
44          stack<TreeNode*>desc;
45          TreeNode* temp2=root;
46        while(temp2){
47            desc.push(temp2);
48            temp2=temp2->right;
49        }
50        // now apply 2 sum logic (find the greatest and smallest element)
51        TreeNode* i=getsmaller(asc,root);
52        TreeNode* j=getbigger(desc,root);
53
54        while(i && j && i!=j){
55            if(i->val+j->val==k){
56                return true;
57            }
58            else if(i->val+j->val >k){
59                j=  getbigger(desc,root);
60            }
61            else{
62              i=  getsmaller(asc,root);
63            }
64        }
65        return false;
66    }
67};
