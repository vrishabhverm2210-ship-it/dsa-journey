2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14bool isNull=false;
15    bool isCompleteTree(TreeNode* root) {
16        queue<TreeNode*>q;
17        q.push(root);
18        while(!q.empty()){
19            TreeNode* node=q.front();
20            q.pop();
21            if(node!=NULL){
22              q.push(node->left);
23              q.push(node->right);
24              if(isNull)return false;   // This condition must be checked in this block only
25            }
26            else{   // means NULL aa gaya
27                    isNull=true;
28                    // q.push(node->left);
29                    // q.push(node->right);
30                }
31            
32        }
33        return true;
34    }
35};
