1class Solution {
2public:
3    int findMaxLength(vector<int>& nums) {
4        unordered_map<int,int>mpp1;
5        int n=nums.size();
6        int diff=0;
7        int onecount=0;
8        int zerocount=0;
9        int res=0;
10        for(int i=0;i<n;i++){
11           if(nums[i]==0)zerocount++;
12           else onecount++;
13           diff=zerocount-onecount;
14           // green condtion(or think ki pura array hi green hai)
15           if(diff==0){res=max(res,i+1); continue;}
16           if(mpp1.find(diff)==mpp1.end()){
17            mpp1[diff]=i;
18           }
19           else{   // mil gya hai diff
20            int idx=mpp1[diff];
21            int maxi=i-idx;
22            res=max(res,maxi);
23           }
24        }
25        return res;
26    }
27};
