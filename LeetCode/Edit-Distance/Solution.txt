3class Solution {
4public:
5    int solveRec(int i, int j, string& word1, string& word2, vector<vector<int>>& dp) {
6        // Base cases
7        if (i == 0) return j; // Need to insert j characters
8        if (j == 0) return i; // Need to delete i characters
9
10        // If already computed
11        if (dp[i][j] != -1) return dp[i][j];
12
13        // If characters match, no operation needed
14        if (word1[i - 1] == word2[j - 1]) {
15            return dp[i][j] = solveRec(i - 1, j - 1, word1, word2, dp);
16        }
17
18        // If characters don't match: try all 3 operations
19        int insertOp = solveRec(i, j - 1, word1, word2, dp);     // Insert
20        int deleteOp = solveRec(i - 1, j, word1, word2, dp);     // Delete
21        int replaceOp = solveRec(i - 1, j - 1, word1, word2, dp); // Replace
22
23        return dp[i][j] = 1 + min({insertOp, deleteOp, replaceOp});
24    }
25
26    int minDistance(string word1, string word2) {
27        int n = word1.length();
28        int m = word2.length();
29
30        // DP table initialized with -1
31        vector<vector<int>> dp(n + 1, vector<int>(m + 1, -1));
32
33        return solveRec(n, m, word1, word2, dp);
34    }
35};
